# Dependencies for the Streamlit app (app.py).
#
# This file is deliberately the small one. Streamlit Community Cloud installs
# whatever it finds in requirements.txt at the repo root, and it used to find
# the whole platform here -- FastAPI, SQLAlchemy, Alembic, pikepdf, pypdfium2,
# cryptography, bcrypt, pytest -- none of which app.py imports. Several of
# those build C++ or Rust from source when no wheel exists for the runtime's
# Python version, which turns a deploy into a compile and often into a failure.
#
# The list below is app.py's actual import closure, nothing more.
#
# For the FastAPI + React platform:   pip install -r requirements-platform.txt
# For S3 or Postgres:                 pip install -r requirements-deploy.txt

pandas>=2.0.0
